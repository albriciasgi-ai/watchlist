     FASE 2: Anti-Throttling (Día 1)

     2.1 Main Process

     // electron/main.js
     const { app, BrowserWindow, powerSaveBlocker } = require('electron');

     // CRÍTICO: Desactivar throttling de Chromium
     app.commandLine.appendSwitch('disable-renderer-backgrounding');
     app.commandLine.appendSwitch('disable-background-timer-throttling');

     app.whenReady().then(() => {
       // Prevenir suspensión del sistema
       powerSaveBlocker.start('prevent-app-suspension');

       const win = new BrowserWindow({
         width: 1400,
         height: 900,
         webPreferences: {
           backgroundThrottling: false,  // CLAVE
           preload: path.join(__dirname, 'preload.js')
         }
       });

       // Dev: localhost:5173 | Prod: archivo local
       win.loadURL(process.env.VITE_DEV_SERVER_URL ||
         `file://${path.join(__dirname, '../dist/index.html')}`);
     });

     Esto resuelve: Gráficos sin gaps al minimizar/inactividad
