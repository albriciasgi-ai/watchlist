     class ZoneDetector:
         """
         Detecta zonas de consolidación (rangos) usando múltiples métodos.
         Permite comparar cuál funciona mejor.
         """

         def detect_zones(self, candles, method, params) -> List[Zone]:
             if method == "pivot_cluster":
                 return self._pivot_cluster_method(candles, params)
             elif method == "atr_based":
                 return self._atr_based_method(candles, params)
             elif method == "volume_profile":
                 return self._volume_profile_method(candles, params)
             elif method == "price_action":
                 return self._price_action_method(candles, params)

     Métodos a implementar y comparar:

     Paso 2: Auto-optimización de parámetros

     Archivo: backend/zone_optimizer.py

     class ZoneOptimizer:
         """
         Prueba combinaciones de parámetros y evalúa calidad de detección.
         """

         def optimize(self, candles, method, param_grid) -> OptimizationResult:
             results = []
             for params in param_grid.combinations():
                 zones = detector.detect_zones(candles, method, params)
                 score = self._evaluate_zones(zones, candles)
                 results.append({params, zones, score})
             return sorted(results, by=score)

         def _evaluate_zones(self, zones, candles) -> float:
             """
             Evalúa qué tan "buenos" son los rangos detectados:
             - ¿El precio realmente rebotó en los extremos?
             - ¿Cuántos toques reales tuvo después de detectarse?
             - ¿El breakout fue limpio o hubo muchos fakeouts?
             """
             score = 0
             for zone in zones:
                 score += self._count_valid_bounces(zone, candles)
                 score += self._evaluate_breakout_quality(zone, candles)
                 score -= self._count_fakeouts(zone, candles)
             return score / len(zones)

     @app.post("/api/zones/test")
     async def test_zone_detection(request: ZoneTestRequest):
         """
         Permite probar detección de zonas con diferentes métodos y parámetros.
         Retorna zonas detectadas para visualización.
         """
         candles = await load_historical(request.symbol, request.interval, request.days)

         if request.optimize:
             # Auto-optimizar y retornar mejor configuración
             results = optimizer.optimize(candles, request.method, request.param_grid)
             return {
                 "best_params": results[0].params,
                 "best_score": results[0].score,
                 "zones": results[0].zones,
                 "all_results": results[:10]  # Top 10
             }
         else:
             # Usar parámetros específicos
             zones = detector.detect_zones(candles, request.method, request.params)
             score = optimizer._evaluate_zones(zones, candles)
             return {"zones": zones, "score": score}

     ---
     Criterios de Evaluación de Zonas

     Para que el auto-optimizador sepa qué es "bueno":

     1. Rebotes Válidos (positivo)

     Después de que se forma la zona, ¿el precio rebotó en los extremos?
     - Toca soporte → sube → +1 punto
     - Toca resistencia → baja → +1 punto

     2. Fakeouts (negativo)

     ¿El precio rompió brevemente y volvió?
     - Rompe resistencia → vuelve dentro → -0.5 puntos
     - Esto indica que la zona no está bien definida

     3. Breakout Limpio (positivo)

     Cuando finalmente rompe, ¿fue decisivo?
     - Rompe con volumen alto → no vuelve → +2 puntos
     - Indica que la zona era real y el breakout es tradeable

     4. Duración de la Zona (neutral/positivo)

     Zonas más largas = más confiables
     - > 24 horas → bonus +1
     - < 4 horas → posible ruido
